    Database rules

    Publishing fails with PERMISSION_DENIED until these are in place. Paste this into Firebase Console → Realtime Database → Rules and hit Publish. It leaves the site publicly readable, lets visitors bump the view counters, and allows writes only to the exact address ziani3618@gmail.com. Change that address in all six places if you ever move accounts. The raw visitor log (IPs and cities) stays readable by you alone.
    First time? Authentication → Sign-in method → Add a provider → Email/Password → Enable, then Users → Add user. Log in here with that e-mail and password.
    Why the address is pinned: your project also has Google sign-in enabled, and a bare auth != null rule would let any Google account on the internet write to your database. Matching on the e-mail closes that. 
    {
  "rules": {
    ".read": false,
    ".write": false,

    "site":        { ".read": true, ".write": "auth != null && auth.token.email == 'ziani3618@gmail.com'" },
    "posts":       { ".read": true, ".write": "auth != null && auth.token.email == 'ziani3618@gmail.com'" },
    "postContent": { ".read": true, ".write": "auth != null && auth.token.email == 'ziani3618@gmail.com'" },

    "reactions": {
      ".read": true,
      "$post": { "$field": {
        ".write": true,
        ".validate": "newData.isNumber() && newData.val() >= 0"
      } }
    },

    "stats": {
      ".read": true,
      ".write": "auth != null && auth.token.email == 'ziani3618@gmail.com'",
      "total":    { ".write": true, ".validate": "newData.isNumber()" },
      "by_year":  { "$k": { ".write": true, ".validate": "newData.isNumber()" } },
      "by_month": { "$k": { ".write": true, ".validate": "newData.isNumber()" } },
      "by_week":  { "$k": { ".write": true, ".validate": "newData.isNumber()" } },
      "by_day":   { "$k": { ".write": true, ".validate": "newData.isNumber()" } }
    },

    "sections": {
      ".read": true,
      ".write": "auth != null && auth.token.email == 'ziani3618@gmail.com'",
      "$page": {
        "total":    { ".write": true, ".validate": "newData.isNumber()" },
        "by_year":  { "$k": { ".write": true, ".validate": "newData.isNumber()" } },
        "by_month": { "$k": { ".write": true, ".validate": "newData.isNumber()" } },
        "by_week":  { "$k": { ".write": true, ".validate": "newData.isNumber()" } },
        "by_day":   { "$k": { ".write": true, ".validate": "newData.isNumber()" } }
      }
    },

    "visitors": {
      ".read": "auth != null && auth.token.email == 'ziani3618@gmail.com'",
      ".write": "auth != null && auth.token.email == 'ziani3618@gmail.com'",
      "$id": { ".write": "auth != null && auth.token.email == 'ziani3618@gmail.com' || !data.exists()" }
    }
  }
}
